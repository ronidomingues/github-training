\begin{frame}{\tit{Git não é GitHub}}
  \vspace{1em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \rox{Git}\\[.4em]
      {\lexendlight Programa de controle de versão. Roda no seu computador,
      sem internet.}\\[.6em]
      \cd{git commit}\\ \cd{git branch}\\ \cd{git merge}
    \end{column}
    \begin{column}{0.48\textwidth}
      \rox{GitHub}\\[.4em]
      {\lexendlight Plataforma que hospeda repositórios Git e acrescenta
      colaboração.}\\[.6em]
      Pull Requests · Issues\\ Pages · Actions · Copilot
    \end{column}
  \end{columns}

  \vspace{1.5em}
  \begin{center}
    \dest{O Git é o motor. O GitHub é a garagem com oficina.}
  \end{center}

  \note{
    \textbf{00:05}\\[.5em]
    A confusão entre os dois é a mais comum. Mostre que o Git funciona
    sozinho: \cd{git init} e \cd{git commit} não precisam de internet.
  }
\end{frame}

\begin{frame}{\tit{Proteja a conta: 2FA}}
  \vspace{0.6em}
  \begin{itemize}
    \item O GitHub \dest{exige} autenticação em dois fatores de quem contribui com código
    \item Aviso por e-mail, \rox{45 dias} para ativar, mais 7 de carência
    \item \rox{Settings} \(\rightarrow\) \rox{Password and authentication} \(\rightarrow\) \rox{Enable 2FA}
    \item Método principal: \dest{aplicativo autenticador} (TOTP)
    \item Reserva: passkey ou chave de segurança
  \end{itemize}

  \vspace{1em}
  \begin{alertblock}{Guarde os códigos de recuperação}
    Eles são a única forma de entrar se você perder o celular.
  \end{alertblock}

  \note{
    \textbf{00:08}\\[.5em]
    \textbf{Faça agora:} quem já tem conta ativa o 2FA em dois minutos.
    Quem está criando, faz depois (guia, capítulo 2).
  }
\end{frame}

\begin{frame}[fragile]{\tit{Dizendo ao Git quem é você}}
\begin{sh}
git config --global user.name "Seu Nome"
git config --global user.email "seu.email@exemplo.com"
git config --global init.defaultBranch main
git config --global pull.rebase false
\end{sh}

  \vspace{0.6em}
  \begin{itemize}
    \item O e-mail deve estar cadastrado na conta do GitHub
    \item Quer privacidade? Use o e-mail \cd{noreply} do GitHub (\rox{Settings} \(\rightarrow\) \rox{Emails})
    \item O \cd{pull.rebase false} evita o erro de \emph{divergent branches} no primeiro conflito
  \end{itemize}

  \note{
    \textbf{00:28}\\[.5em]
    Peça para todos conferirem com \cd{git config --global --list}.
  }
\end{frame}

\begin{frame}[fragile]{\tit{Login: o gh resolve}}

\end{sh}

\begin{frame}[fragile]{\tit{O primeiro repositório}}
\begin{sh}
gh repo create meu-primeiro-repo --public --clone
cd meu-primeiro-repo
echo "# Meu Primeiro Projeto" > README.md
git status
git add README.md
git commit -m "docs: cria o README"
git push -u origin main
\end{sh}

  \vspace{0.3em}
  \begin{center}
    \rox{O \cd{-u} liga a sua main à do GitHub: depois, basta \cd{git push}.}
  \end{center}

\begin{frame}[fragile]{\tit{Branches}}
\begin{sh}
git switch -c feature/login      # cria a branch e entra nela
# ... edita, add, commit ...
git switch main                  # volta para a main
git merge feature/login          # traz o trabalho
\end{sh}

  \vspace{0.5em}
  \begin{itemize}
    \item Branch é uma \dest{linha de trabalho}, não uma cópia do projeto
    \item Criar uma branch é instantâneo, mesmo num projeto enorme
    \item \cd{git switch} é o comando moderno; o \cd{checkout} antigo ainda funciona
  \end{itemize}

  \note{
    \textbf{00:50}\\[.5em]
    Desenhe no quadro: a main como uma linha, a branch saindo dela e voltando
    no merge.
  }
\end{frame}

\begin{frame}[fragile]{\tit{Conflito não é erro}}
  \vspace{0.2em}
  {\small Duas pessoas mudaram a \textbf{mesma linha}. O Git pede uma decisão:}
\begin{sh}
<<<<<<< HEAD
# Meu Primeiro Projeto (editado no computador)
=======
# Meu Primeiro Projeto (editado no site)
>>>>>>> 15261e4
\end{sh}
  \begin{enumerate}
    \item Deixe só o que deve ficar e apague os marcadores
    \item \cd{git add README.md}
    \item \cd{git commit} e \cd{git push}
  \end{enumerate}
  \begin{center}
    \rox{Perdeu-se? \cd{git merge --abort} volta ao estado de antes.}
  \end{center}

  \note{
    \textbf{00:54}\\[.5em]
    \textbf{Faça agora:} Exercício 4 do guia. Ele cria o conflito de propósito:
    edita a mesma linha no site e no computador.
  }
\end{frame}

\begin{frame}[fragile]{\tit{GitHub Actions em um arquivo}}
\begin{yml}
on:
  push:
    branches: [ main ]
permissions:
  contents: write            # sem isto, o git push dá 403
jobs:
  executar:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
      - uses: actions/setup-python@v7
      - run: python main.py
\end{yml}
  {\footnotesize \textcolor{codecomment}{\texttt{.github/workflows/python.yml} — exemplo completo no guia, capítulo 8}}

  \note{
    \textbf{01:32}\\[.5em]
    Vocabulário: evento, job, step, runner. E o mito do \cd{[skip ci]}: o
    push feito pelo próprio workflow com o GITHUB\_TOKEN \textbf{não} dispara
    o workflow de novo.
  }
\end{frame}

\begin{frame}[fragile]{\tit{Git LFS: arquivos grandes}}
\begin{sh}
git lfs install                  # uma vez por usuário
git lfs track "*.psd"            # grava a regra no .gitattributes
git add .gitattributes arte.psd
git commit -m "chore: versiona a arte com Git LFS"
\end{sh}
  \begin{itemize}
    \item O GitHub \dest{recusa arquivos acima de 100 MiB} num push comum
    \item No repositório fica só um ponteiro; o arquivo vai para o LFS
    \item 10 GiB de armazenamento e 10 GiB de banda por mês
  \end{itemize}

  \note{
    \textbf{01:40}\\[.5em]
    O limite de 100 MiB é do GitHub, não do Git. Configure o LFS antes de
    adicionar o arquivo grande.
  }
\end{frame}
